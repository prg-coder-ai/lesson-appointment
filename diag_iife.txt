(function () {
      try {
        var sp = new URLSearchParams(window.location.search);
        // 兼容「?tCode =tenant_A」（参数名带空格）：原生 get 对空格键名返回 null，故手动按 '&' 解析
        var tCode = (function () {
          var pairs = (window.location.search || '').replace(/^\?/, '').split('&');
          for (var i = 0; i < pairs.length; i++) {
            if (!pairs[i]) continue;
            var eq = pairs[i].indexOf('=');
            var k = eq === -1 ? pairs[i] : pairs[i].slice(0, eq);
            var v = eq === -1 ? '' : pairs[i].slice(eq + 1);
            if (decodeURIComponent(k).trim().toLowerCase() === 'tcode') return decodeURIComponent(v);
          }
          return '';
        })();
        if (!tCode) tCode = sp.get('tcode') || '';
        if (!tCode) {
          // 如果没有 tCode 参数，尝试从 localStorage 读取上次注册/登录的租户编码
          tCode = localStorage.getItem('lastRegisterTenantCode') || localStorage.getItem('lastLoginTenantCode') || '';
        }
        tCode = window.sanitizeTenantCode(tCode);
        var noRedirect = sp.get('noredirect');
        var hasToken = !!localStorage.getItem('token');
        if (tCode && !hasToken && noRedirect !== '1') {
          window.location.replace('student-landing.html?tCode=' + encodeURIComponent(tCode) + '&from=index');
        }
      } catch (e) {}
    })
