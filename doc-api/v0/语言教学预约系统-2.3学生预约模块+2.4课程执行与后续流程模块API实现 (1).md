```java
package com.language.reservation.entity;

import lombok.Data;
import javax.validation.constraints.NotBlank;
import javax.validation.constraints.NotNull;
import java.math.BigDecimal;
import java.util.Date;

/**
 * 预约订单实体类，对应设计2.3 学生预约模块-预约下单、支付、取消等接口
 * 关联排期（scheduleId）、学生（studentId）、课程（courseId）

@Data

    private String orderId;  // 唯一标识（UUID），对应通用校验规则-ID类参数

    private String payNo;       // 支付流水号（支付宝返回）
```

## 1.2 控制器（StudentReservationController.java）

```java
package com.language.reservation.controller;

import com.language.reservation.common.Result;
import com.language.reservation.entity.ReservationOrder;
import com.language.reservation.entity.ReservationQuery;
import com.language.reservation.service.StudentReservationService;
import com.language.reservation.utils.PermissionCheck;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.validation.annotation.Validated;
import org.springframework.web.bind.annotation.*;

import javax.servlet.http.HttpServletRequest;
import java.util.List;
import java.util.Map;

/**
 * 学生预约模块控制器，对应设计2.3 学生预约模块所有接口
 * 接口权限：仅学生可操作（通过PermissionCheck校验）
 */
@RestController
@RequestMapping("/reservation")
@Validated
public class StudentReservationController {

    @Autowired
    private StudentReservationService reservationService;
    @Autowired
    private PermissionCheck permissionCheck;

    /**
     * 预约下单接口，对应设计2.3 接口：/api/v1/reservation/create（学生权限）
     * 功能：学生选择排期创建预约订单，生成待支付订单
     */
    @PostMapping("/create")
    public Result<Map<String, String>> createReservation(
            @Validated @RequestBody ReservationOrder order,
            @RequestHeader("Authorization") String token) {
        // 权限校验：仅学生可操作
        permissionCheck.checkStudent(token);
        // 校验订单归属当前学生（确保学生只能创建自己的订单）
        String studentId = permissionCheck.getUserIdFromToken(token);
        order.setStudentId(studentId);
        // 调用服务层创建订单
        Map<String, String> resultMap = reservationService.createReservation(order);
        return Result.success(resultMap, "预约订单创建成功，请尽快支付");
    }

    /**
     * 订单支付接口，对应设计2.3 接口：/api/v1/reservation/pay（学生权限）
     * 功能：学生支付待支付订单，对接支付宝支付
     */
    @PostMapping("/pay")
    public Result<Map<String, String>> payOrder(
            @NotBlank(message = "订单ID不能为空") String orderId,
            @RequestHeader("Authorization") String token) {
        // 权限校验：仅学生可操作
        permissionCheck.checkStudent(token);
        // 校验订单归属当前学生
        String studentId = permissionCheck.getUserIdFromToken(token);
        permissionCheck.checkStudentOwner(token, studentId);
        // 调用服务层发起支付，返回支付宝支付页面链接
        Map<String, String> payInfo = reservationService.payOrder(orderId, studentId);
        return Result.success(payInfo, "支付链接生成成功");
    }

    /**
     * 支付同步回调接口，对应设计2.3 支付功能-同步回调
     * 功能：支付宝支付成功后跳转回调，更新订单状态
     */
    @GetMapping("/pay/return")
    public Result<Map<String, String>> payReturn(HttpServletRequest request) {
        // 解析支付宝回调参数，更新订单状态
        Map<String, String> resultMap = reservationService.handlePayReturn(request);
        return Result.success(resultMap, "支付成功");
    }

    /**
     * 支付异步回调接口，对应设计2.3 支付功能-异步回调
     * 功能：支付宝后台异步通知，确保订单状态同步（防止前端未回调）
     */
    @PostMapping("/pay/notify")
    public String payNotify(HttpServletRequest request) {
        // 处理支付宝异步通知，更新订单状态，返回success告知支付宝已接收
        boolean handleSuccess = reservationService.handlePayNotify(request);
        return handleSuccess ? "success" : "fail";
    }

    /**
     * 取消预约接口，对应设计2.3 接口：/api/v1/reservation/cancel（学生权限）
     * 功能：学生取消未支付订单，或符合条件的已支付订单（未到课）
     */
    @PostMapping("/cancel")
    public Result<Void> cancelReservation(
            @NotBlank(message = "订单ID不能为空") String orderId,
            @RequestHeader("Authorization") String token) {
        // 权限校验：仅学生可操作
        permissionCheck.checkStudent(token);
        // 校验订单归属当前学生
        String studentId = permissionCheck.getUserIdFromToken(token);
        reservationService.checkOrderOwner(orderId, studentId);
        // 调用服务层取消订单
        reservationService.cancelReservation(orderId);
        return Result.success(null, "预约订单取消成功");
    }

    /**
     * 预约查询接口，对应设计2.3 接口：/api/v1/reservation/list（学生权限）
     * 功能：学生查询自己的所有预约订单，支持多条件筛选
     */
    @GetMapping("/list")
    public Result<Map<String, Object>> getReservationList(
            ReservationQuery query,
            @RequestHeader("Authorization") String token) {
        // 权限校验：仅学生可操作
        permissionCheck.checkStudent(token);
        // 设置当前学生ID为查询条件（默认查询自己的订单）
        String studentId = permissionCheck.getUserIdFromToken(token);
        query.setStudentId(studentId);
        // 调用服务层查询订单列表（含分页）
        Map<String, Object> resultMap = reservationService.getReservationList(query);
        return Result.success(resultMap, "订单查询成功");
    }

    /**
     * 订单详情接口，对应设计2.3 接口：/api/v1/reservation/detail（学生权限）
     * 功能：学生查看单个预约订单的详细信息
     */
    @GetMapping("/detail")
    public Result<ReservationOrder> getReservationDetail(
            @NotBlank(message = "订单ID不能为空") String orderId,
            @RequestHeader("Authorization") String token) {
        // 权限校验：仅学生可操作
        permissionCheck.checkStudent(token);
        // 校验订单归属当前学生
        String studentId = permissionCheck.getUserIdFromToken(token);
        reservationService.checkOrderOwner(orderId, studentId);
        // 调用服务层查询订单详情
        ReservationOrder order = reservationService.getReservationDetail(orderId);
        return Result.success(order, "订单详情查询成功");
    }
}
```

## 1.3 服务层（StudentReservationService.java）

```java
package com.language.reservation.service;

import com.alipay.api.AlipayApiException;
import com.alipay.api.AlipayClient;
import com.alipay.api.DefaultAlipayClient;
import com.alipay.api.request.AlipayTradePagePayRequest;
import com.language.reservation.entity.Course;
import com.language.reservation.entity.ReservationOrder;
import com.language.reservation.entity.ReservationQuery;
import com.language.reservation.entity.Schedule;
import com.language.reservation.exception.BusinessException;
import com.language.reservation.exception.ResourceNotFoundException;
import com.language.reservation.mapper.CourseMapper;
import com.language.reservation.mapper.ReservationOrderMapper;
import com.language.reservation.mapper.ScheduleMapper;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import javax.servlet.http.HttpServletRequest;
import java.text.ParseException;
import java.text.SimpleDateFormat;
import java.util.Date;
import java.util.HashMap;
import java.util.List;
import java.util.Map;
import java.util.UUID;

/**
 * 学生预约模块服务层，对应设计2.3 学生预约模块所有接口的业务逻辑
 * 涵盖预约下单、支付、取消、查询等功能，严格遵循权限校验和业务规则
 */
@Service
public class StudentReservationService {

    @Autowired
    private ReservationOrderMapper orderMapper;
    @Autowired
    private ScheduleMapper scheduleMapper;
    @Autowired
    private CourseMapper courseMapper;

    // 支付宝配置（从application.yml读取，对应设计2.3 支付功能）
    @Value("${alipay.app-id}")
    private String alipayAppId;
    @Value("${alipay.private-key}")
    private String alipayPrivateKey;
    @Value("${alipay.public-key}")
    private String alipayPublicKey;
    @Value("${alipay.gateway-url}")
    private String alipayGatewayUrl;
    @Value("${alipay.return-url}")
    private String alipayReturnUrl;
    @Value("${alipay.notify-url}")
    private String alipayNotifyUrl;

    // 日期格式化工具，对应通用校验规则-时间格式
    private static final SimpleDateFormat DATE_FORMAT = new SimpleDateFormat("yyyy-MM-dd HH:mm:ss");

    /**
     * 预约下单，对应设计2.3 预约下单接口
     */
    @Transactional
    public Map<String, String> createReservation(ReservationOrder order) {
        // 1. 校验排期是否存在且可预约
        Schedule schedule = scheduleMapper.selectScheduleById(order.getScheduleId());
        if (schedule == null) {
            throw new ResourceNotFoundException("排期不存在，无法预约");
        }
        if (!"available".equals(schedule.getStatus())) {
            throw new BusinessException("该排期不可预约（已被占用或已取消）");
        }

        // 2. 校验课程是否存在
        Course course = courseMapper.selectCourseById(order.getCourseId());
        if (course == null) {
            throw new ResourceNotFoundException("课程不存在，无法预约");
        }

        // 3. 校验排期与课程关联关系（防止非法预约）
        if (!schedule.getCourseId().equals(order.getCourseId())) {
            throw new BusinessException("排期与课程不匹配，无法预约");
        }

        // 4. 校验学生是否已预约该排期（避免重复预约，对应设计2.3 预约规则）
        if (orderMapper.selectByStudentAndSchedule(order.getStudentId(), order.getScheduleId()) != null) {
            throw new BusinessException("您已预约该排期，请勿重复预约");
        }

        // 5. 生成订单ID，设置订单默认状态（待支付）
        String orderId = UUID.randomUUID().toString();
        order.setOrderId(orderId);
        order.setOrderStatus("pending");
        order.setCreateTime(new Date());

        // 6. 插入订单到数据库
        orderMapper.insertOrder(order);

        // 7. 组装返回数据（订单ID、支付提示）
        Map<String, String> resultMap = new HashMap<>();
        resultMap.put("orderId", orderId);
        resultMap.put("tips", "订单创建成功，30分钟内未支付将自动取消");
        return resultMap;
    }

    /**
     * 订单支付，对接支付宝，对应设计2.3 支付功能
     */
    public Map<String, String> payOrder(String orderId, String studentId) {
        // 1. 校验订单是否存在
        ReservationOrder order = orderMapper.selectOrderById(orderId);
        if (order == null) {
            throw new ResourceNotFoundException("订单不存在，无法支付");
        }

        // 2. 校验订单归属当前学生
        if (!order.getStudentId().equals(studentId)) {
            throw new BusinessException("您无权支付他人订单");
        }

        // 3. 校验订单状态（仅待支付订单可支付）
        if (!"pending".equals(order.getOrderStatus())) {
            throw new BusinessException("订单状态异常，仅待支付订单可支付");
        }

        // 4. 校验订单支付超时（30分钟超时，对应设计2.3 预约规则）
        long createTime = order.getCreateTime().getTime();
        long currentTime = System.currentTimeMillis();
        if (currentTime - createTime > 30 * 60 * 1000) {
            // 超时自动取消订单
            order.setOrderStatus("cancelled");
            order.setCancelTime(DATE_FORMAT.format(new Date()));
            orderMapper.updateOrderStatus(order);
            throw new BusinessException("订单已超时，已自动取消，请重新预约");
        }

        // 5. 对接支付宝，生成支付链接
        AlipayClient alipayClient = new DefaultAlipayClient(
                alipayGatewayUrl, alipayAppId, alipayPrivateKey, "json", "UTF-8", alipayPublicKey, "RSA2");
        AlipayTradePagePayRequest request = new AlipayTradePagePayRequest();
        // 设置回调地址
        request.setReturnUrl(alipayReturnUrl);
        request.setNotifyUrl(alipayNotifyUrl);

        // 组装支付参数
        Map<String, String> bizContent = new HashMap<>();
        bizContent.put("out_trade_no", orderId); // 订单号（系统生成）
        bizContent.put("total_amount", order.getOrderAmount().toString()); // 订单金额
        bizContent.put("subject", "语言教学预约-" + order.getCourseId()); // 订单标题
        bizContent.put("product_code", "FAST_INSTANT_TRADE_PAY"); // 支付宝产品码

        request.setBizContent(new com.alibaba.fastjson.JSONObject(bizContent).toString());

        try {
            // 调用支付宝接口，获取支付页面链接
            String form = alipayClient.pageExecute(request).getBody();
            Map<String, String> resultMap = new HashMap<>();
            resultMap.put("payForm", form); // 支付表单（前端渲染即可跳转支付宝）
            return resultMap;
        } catch (AlipayApiException e) {
            e.printStackTrace();
            throw new BusinessException("支付链接生成失败，请稍后再试");
        }
    }

    /**
     * 支付同步回调处理，对应设计2.3 支付功能-同步回调
     */
    @Transactional
    public Map<String, String> handlePayReturn(HttpServletRequest request) {
        // 1. 解析支付宝回调参数（简化处理，实际需校验签名）
        String orderId = request.getParameter("out_trade_no");
        String tradeNo = request.getParameter("trade_no"); // 支付宝流水号
        String tradeStatus = request.getParameter("trade_status"); // 支付状态

        // 2. 校验支付状态（TRADE_SUCCESS为支付成功）
        if (!"TRADE_SUCCESS".equals(tradeStatus)) {
            throw new BusinessException("支付未成功，请核对支付状态");
        }

        // 3. 校验订单是否存在
        ReservationOrder order = orderMapper.selectOrderById(orderId);
        if (order == null) {
            throw new ResourceNotFoundException("订单不存在，支付回调失败");
        }

        // 4. 更新订单状态为已支付
        order.setOrderStatus("paid");
        order.setPayTime(DATE_FORMAT.format(new Date()));
        order.setPayNo(tradeNo);
        orderMapper.updateOrderStatus(order);

        // 5. 更新排期状态为不可预约（避免重复预约）
        Schedule schedule = scheduleMapper.selectScheduleById(order.getScheduleId());
        schedule.setStatus("unavailable");
        scheduleMapper.updateScheduleStatus(schedule);

        // 6. 组装返回数据
        Map<String, String> resultMap = new HashMap<>();
        resultMap.put("orderId", orderId);
        resultMap.put("tradeNo", tradeNo);
        resultMap.put("message", "支付成功，您的预约已确认");
        return resultMap;
    }

    /**
     * 支付异步回调处理，对应设计2.3 支付功能-异步回调
     */
    @Transactional
    public boolean handlePayNotify(HttpServletRequest request) {
        try {
            // 1. 解析支付宝回调参数（简化处理，实际需校验签名）
            String orderId = request.getParameter("out_trade_no");
            String tradeNo = request.getParameter("trade_no");
            String tradeStatus = request.getParameter("trade_status");

            // 2. 校验支付状态和订单
            if (!"TRADE_SUCCESS".equals(tradeStatus)) {
                return false;
            }
            ReservationOrder order = orderMapper.selectOrderById(orderId);
            if (order == null || !"pending".equals(order.getOrderStatus())) {
                return false;
            }

            // 3. 更新订单状态（与同步回调逻辑一致，防止前端未回调）
            order.setOrderStatus("paid");
            order.setPayTime(DATE_FORMAT.format(new Date()));
            order.setPayNo(tradeNo);
            orderMapper.updateOrderStatus(order);

            // 4. 更新排期状态
            Schedule schedule = scheduleMapper.selectScheduleById(order.getScheduleId());
            schedule.setStatus("unavailable");
            scheduleMapper.updateScheduleStatus(schedule);

            return true;
        } catch (Exception e) {
            e.printStackTrace();
            return false;
        }
    }

    /**
     * 取消预约订单，对应设计2.3 取消预约接口
     */
    @Transactional
    public void cancelReservation(String orderId) {
        // 1. 校验订单是否存在
        ReservationOrder order = orderMapper.selectOrderById(orderId);
        if (order == null) {
            throw new ResourceNotFoundException("订单不存在，无法取消");
        }

        // 2. 校验订单状态（仅待支付、已支付且未到课的订单可取消）
        if ("cancelled".equals(order.getOrderStatus()) || "refunded".equals(order.getOrderStatus())) {
            throw new BusinessException("订单已取消或已退款，无需重复操作");
        }

        // 3. 已支付订单取消：校验是否已到课（到课不可取消）
        if ("paid".equals(order.getOrderStatus())) {
            Schedule schedule = scheduleMapper.selectScheduleById(order.getScheduleId());
            try {
                Date startTime = DATE_FORMAT.parse(schedule.getStartTime());
                Date currentTime = new Date();
                if (currentTime.after(startTime)) {
                    throw new BusinessException("课程已开始，无法取消预约");
                }
                // 已支付取消，后续可添加退款逻辑（对应设计2.3 退款规则）
                order.setOrderStatus("refunded");
                order.setRefundTime(DATE_FORMAT.format(new Date()));
                // 恢复排期可预约状态
                schedule.setStatus("available");
                scheduleMapper.updateScheduleStatus(schedule);
            } catch (ParseException e) {
                throw new BusinessException("时间格式错误，无法取消预约");
            }
        } else {
            // 待支付订单取消
            order.setOrderStatus("cancelled");
        }

        // 4. 更新订单取消时间和状态
        order.setCancelTime(DATE_FORMAT.format(new Date()));
        orderMapper.updateOrderStatus(order);
    }

    /**
     * 查询预约订单列表，对应设计2.3 预约查询接口
     */
    public Map<String, Object> getReservationList(ReservationQuery query) {
        // 1. 查询订单列表（支持多条件筛选）
        List<ReservationOrder> orderList = orderMapper.selectOrderList(query);
        // 2. 查询总条数（分页用）
        int total = orderMapper.selectOrderCount(query);
        // 3. 组装返回数据（列表+总条数）
        Map<String, Object> resultMap = new HashMap<>();
        resultMap.put("orderList", orderList);
        resultMap.put("total", total);
        return resultMap;
    }

    /**
     * 查询订单详情，对应设计2.3 订单详情接口
     */
    public ReservationOrder getReservationDetail(String orderId) {
        ReservationOrder order = orderMapper.selectOrderById(orderId);
        if (order == null) {
            throw new ResourceNotFoundException("订单不存在");
        }
        return order;
    }

    /**
     * 校验订单归属（辅助方法），对应设计2.3 权限控制
     */
    public void checkOrderOwner(String orderId, String studentId) {
        ReservationOrder order = orderMapper.selectOrderById(orderId);
        if (order == null) {
            throw new ResourceNotFoundException("订单不存在");
        }
        if (!order.getStudentId().equals(studentId)) {
            throw new BusinessException("您无权操作他人订单");
        }
    }
}
```

```java
import com.language.reservation.entity.ReservationOrder;

import org.apache.ibatis.annotations.Mapper;
import org.apache.ibatis.annotations.Param;

import java.util.List;
```
